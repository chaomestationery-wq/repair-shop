```javascript /** ร้านซ่อมมือถือ – Backend Google Apps Script + Google Sheets * Script Properties: * LINE_TOKEN = LINE Messaging API Channel Access Token * OWNER_ID = LINE User ID ของเจ้าของร้าน */ const P = PropertiesService.getScriptProperties(); const STATUS = [ 'รับเรื่องแล้ว', 'กำลังไปรับเครื่อง', 'รับเครื่องแล้ว', 'กำลังซ่อม', 'ซ่อมเสร็จ', 'กำลังนำส่ง', 'ส่งมอบแล้ว' ]; const NOTIFY = [ 'กำลังไปรับเครื่อง', 'ซ่อมเสร็จ', 'กำลังนำส่ง', 'ส่งมอบแล้ว' ]; const HEAD = [ 'รหัส', 'เวลา', 'ชื่อ', 'เบอร์', 'เครื่อง/อาการ', 'ที่อยู่', 'lat', 'lng', 'แผนที่', 'userId', 'สถานะ', 'history' ]; const STATUS_COL = 11; const HISTORY_COL = 12; const SHEET_NAME = 'Orders'; /* -------------------- พื้นฐาน -------------------- */ function sheet_() { const ss = SpreadsheetApp.getActive(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.getRange(1, 1, 1, HEAD.length).setValues([HEAD]); sh.setFrozenRows(1); } return sh; } function out_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function clean_(value, max) { return String(value == null ? if (!owner) { throw new Error('กรุณาตั้งค่า OWNER_ID ก่อน'); } push_(owner, 'ทดสอบระบบแจ้งเตือนร้านซ่อมมือถือ'); console.log('ส่งคำสั่งทดสอบแล้ว กรุณาตรวจสอบ LINE'); } /* -------------------- ตรวจสอบข้อมูล -------------------- */ /** * ดูจำนวนออเดอร์และจำนวนออเดอร์ที่มี userId */ function checkOrders() { const sh = sheet_(); const lastRow = sh.getLastRow(); if (lastRow < 2) { console.log('ยังไม่มีออเดอร์'); return; } const rows = sh .getRange(2, 1, lastRow - 1, HEAD.length) .getValues(); const total = rows.length; const withUser = rows.filter(function (r) { return clean_(r[9], 100) !== ''; }).length; console.log('ออเดอร์ทั้งหมด: ' + total); console.log('ออเดอร์ที่มี userId: ' + withUser); console.log('ออเดอร์ที่ไม่มี userId: ' + (total - withUser)); } ```
